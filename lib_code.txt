class Medication {
  final String name;
  final String dosage;
  final String quantity;
  final String remainingQuantity;
  final String instructions;
  final String notes;
  final List<String> reminderTimes;
  final String doseStatus;
  final String doseStatusDate;
  final Map<String, String> doseRecords;
  final String startDate;
  final String endDate;

  const Medication({
    required this.name,
    required this.dosage,
    this.quantity = "",
    String? remainingQuantity,
    required this.instructions,
    this.notes = "",
    this.reminderTimes = const [],
    this.doseStatus = "notTakenYet",
    this.doseStatusDate = "",
    this.doseRecords = const {},
    this.startDate = "",
    this.endDate = "",
  }) : remainingQuantity = remainingQuantity ?? quantity;

  Medication copyWith({
    String? name,
    String? dosage,
    String? quantity,
    String? remainingQuantity,
    String? instructions,
    String? notes,
    List<String>? reminderTimes,
    String? doseStatus,
    String? doseStatusDate,
    Map<String, String>? doseRecords,
    String? startDate,
    String? endDate,
  }) {
    return Medication(
      name: name ?? this.name,
      dosage: dosage ?? this.dosage,
      quantity: quantity ?? this.quantity,
      remainingQuantity: remainingQuantity ?? this.remainingQuantity,
      instructions: instructions ?? this.instructions,
      notes: notes ?? this.notes,
      reminderTimes: reminderTimes ?? this.reminderTimes,
      doseStatus: doseStatus ?? this.doseStatus,
      doseStatusDate: doseStatusDate ?? this.doseStatusDate,
      doseRecords: doseRecords ?? this.doseRecords,
      startDate: startDate ?? this.startDate,
      endDate: endDate ?? this.endDate,
    );
  }

  String getDoseRecord(String key) {
    return doseRecords[key] ?? "";
  }

  Map<String, dynamic> toJson() {
    return {
      "name": name,
      "dosage": dosage,
      "quantity": quantity,
      "remainingQuantity": remainingQuantity,
      "instructions": instructions,
      "notes": notes,
      "reminderTimes": reminderTimes,
      "doseStatus": doseStatus,
      "doseStatusDate": doseStatusDate,
      "doseRecords": doseRecords,
      "startDate": startDate,
      "endDate": endDate,
    };
  }

  factory Medication.fromJson(Map<String, dynamic> json) {
    Map<String, String> parsedDoseRecords = {};

    final rawDoseRecords = json["doseRecords"];

    if (rawDoseRecords is Map) {
      parsedDoseRecords = rawDoseRecords.map(
        (key, value) => MapEntry(
          key.toString(),
          value.toString(),
        ),
      );
    }

    final quantityValue = (json["quantity"] ?? "").toString();
    final remainingQuantityValue =
        (json["remainingQuantity"] ?? quantityValue).toString();

    return Medication(
      name: json["name"] ?? "",
      dosage: json["dosage"] ?? "",
      quantity: quantityValue,
      remainingQuantity: remainingQuantityValue,
      instructions: json["instructions"] ?? "",
      notes: json["notes"] ?? "",
      reminderTimes: List<String>.from(json["reminderTimes"] ?? []),
      doseStatus: json["doseStatus"] ?? "notTakenYet",
      doseStatusDate: json["doseStatusDate"] ?? "",
      doseRecords: parsedDoseRecords,
      startDate: json["startDate"] ?? "",
      endDate: json["endDate"] ?? "",
    );
  }

import 'app_language.dart';

class ReminderPage extends StatefulWidget {
  final Medication medication;
  final int medicationIndex;

  const ReminderPage({
    super.key,
    required this.medication,
    this.medicationIndex = -1,
  });

  @override
  State<ReminderPage> createState() => _ReminderPageState();
}

class _ReminderPageState extends State<ReminderPage> {
  late Medication medication;

  bool isSaving = false;

  static const int doseWindowMinutes = 120;

  @override
  void initState() {
    super.initState();
    medication = widget.medication;

    WidgetsBinding.instance.addPostFrameCallback((_) async {
      await TimeHelper.loadSavedTimeFormat();

      if (mounted) {
        setState(() {});
      }

      await autoAdjustDefaultReminderTime();
      await autoMarkPastDosesMissed();
    });
  }

  bool shouldAutoReplaceDefaultReminderTime() {
    if (medication.reminderTimes.length != 1) {
      return false;
    }

    if (medication.reminderTimes.first != "08:00") {
      return false;
    }

    final generatedTimes = TimeHelper.generateReminderTimesFromInstructions(
      medication.instructions,
    );

    if (generatedTimes.length != 1) {
      return true;
    }

    return TimeHelper.timeToString(generatedTimes.first) != "08:00";
  }

  Future<void> autoAdjustDefaultReminderTime() async {
    if (!shouldAutoReplaceDefaultReminderTime()) {
      return;
    }

    final generatedTimes = TimeHelper.generateReminderTimesFromInstructions(
      medication.instructions,
    ).map((time) => TimeHelper.timeToString(time)).toList();

    if (generatedTimes.isEmpty) {
      return;
    }

    final updatedMedication = medication.copyWith(
      reminderTimes: generatedTimes,
      doseRecords: const {},
      doseStatus: "notTakenYet",
      doseStatusDate: "",
    );

    setState(() {
      medication = updatedMedication;
    });

    await saveMedicationChanges();
  }

  List<TimeOfDay> get reminderTimes {
    if (medication.reminderTimes.isNotEmpty) {
      return medication.reminderTimes
          .map((time) => TimeHelper.stringToTime(time))
          .toList();
    }

    return TimeHelper.generateReminderTimesFromInstructions(
      medication.instructions,
    );
  }

  String doseRecordKey(TimeOfDay time) {
    return "${DateHelper.todayString()}|${TimeHelper.timeToString(time)}";
  }

  String getDoseStatus(TimeOfDay time) {
    final key = doseRecordKey(time);
    final savedStatus = medication.doseRecords[key];

    if (savedStatus == "taken" || savedStatus == "missed") {
      return savedStatus!;
    }

    final now = DateTime.now();
    final scheduledDateTime = TimeHelper.getDateTimeForToday(time);
    final endOfWindow = scheduledDateTime.add(
      const Duration(minutes: doseWindowMinutes),
    );

    if (now.isBefore(scheduledDateTime)) {
      return "upcoming";
    }

    if (now.isAfter(endOfWindow)) {
      return "missed";
    }

    return "due";
  }

  Future<void> autoMarkPastDosesMissed() async {
    final updatedRecords = Map<String, String>.from(medication.doseRecords);
    bool changed = false;

    for (final time in reminderTimes) {
      final key = doseRecordKey(time);

      if (updatedRecords.containsKey(key)) {
        continue;
      }

      final now = DateTime.now();
      final scheduledDateTime = TimeHelper.getDateTimeForToday(time);
      final endOfWindow = scheduledDateTime.add(
        const Duration(minutes: doseWindowMinutes),
      );

      if (now.isAfter(endOfWindow)) {
        updatedRecords[key] = "missed";
        changed = true;
      }
    }

    if (!changed) {
      return;
    }

    final updatedMedication = medication.copyWith(
      doseRecords: updatedRecords,
      doseStatus: getMainDoseStatus(updatedRecords),
      doseStatusDate: DateHelper.todayString(),
    );

    setState(() {
      medication = updatedMedication;
    });

    await saveMedicationChanges();
  }

  String getMainDoseStatus(Map<String, String> records) {
    final today = DateHelper.todayString();
    final times = reminderTimes;

    int takenCount = 0;
    int missedCount = 0;

    for (final time in times) {
      final key = "$today|${TimeHelper.timeToString(time)}";
      final status = records[key];

      if (status == "taken") {
        takenCount++;
      }

      if (status == "missed") {
        missedCount++;
      }
    }

    if (times.isNotEmpty && takenCount == times.length) {
      return "taken";
    }

    if (missedCount > 0) {
      return "missed";
    }

    return "notTakenYet";
  }

  Future<void> saveMedicationChanges() async {
    if (widget.medicationIndex < 0) {
      return;
    }

    await MedicationStorage.updateMedication(
      widget.medicationIndex,
      medication,
    );
  }

  int parseQuantityNumber(String value) {
    final match = RegExp(r'\d+').firstMatch(value);

    if (match == null) {
      return 0;
    }

    return int.tryParse(match.group(0) ?? "") ?? 0;
  }

  String decreaseRemainingQuantityByOne() {
    final currentRemaining = parseQuantityNumber(medication.remainingQuantity);

    if (currentRemaining <= 0) {
      return medication.remainingQuantity;
    }

    return (currentRemaining - 1).toString();
  }

  String increaseRemainingQuantityByOne() {
    final currentRemaining = parseQuantityNumber(medication.remainingQuantity);
    final totalQuantity = parseQuantityNumber(medication.quantity);

    if (currentRemaining <= 0 && totalQuantity <= 0) {
      return medication.remainingQuantity;
    }

    if (totalQuantity > 0 && currentRemaining >= totalQuantity) {
      return totalQuantity.toString();
    }

    return (currentRemaining + 1).toString();
  }

  Future<void> markDose(TimeOfDay time, String status) async {
    if (isSaving) {
      return;
    }

    final key = doseRecordKey(time);
    final previousSavedStatus = medication.doseRecords[key] ?? "";
    final currentStatus = getDoseStatus(time);

    if (status == "taken" && currentStatus != "due") {
      showNotAvailableMessage();
      return;
    }

    if (status == "missed" &&
        currentStatus != "due" &&
        currentStatus != "upcoming") {
      showNotAvailableMessage();
      return;
    }

    setState(() {
      isSaving = true;
    });

    final updatedRecords = Map<String, String>.from(medication.doseRecords);
    updatedRecords[key] = status;

    String updatedRemainingQuantity = medication.remainingQuantity;

    if (status == "taken" && previousSavedStatus != "taken") {
      updatedRemainingQuantity = decreaseRemainingQuantityByOne();
    }

    if (previousSavedStatus == "taken" && status != "taken") {
      updatedRemainingQuantity = increaseRemainingQuantityByOne();
    }

    final updatedMedication = medication.copyWith(
      remainingQuantity: updatedRemainingQuantity,
      doseRecords: updatedRecords,
      doseStatus: getMainDoseStatus(updatedRecords),
      doseStatusDate: DateHelper.todayString(),
    );

    try {
      setState(() {
        medication = updatedMedication;
      });

      await saveMedicationChanges();
    } finally {
      if (mounted) {
        setState(() {
          isSaving = false;
        });
      }
    }
  }

  void showNotAvailableMessage() {
    final language = AppLanguage.currentLanguage.value;

    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(
        content: Text(
          language == "en"
              ? "This dose cannot be marked as taken now."
              : "Không thể đánh dấu liều này là đã uống vào lúc này.",
        ),
      ),
    );
  }

  Future<TimeOfDay?> pickCustomerTime(TimeOfDay initialTime) async {
    return showTimePicker(
      context: context,
      initialTime: initialTime,
      initialEntryMode: TimePickerEntryMode.input,
      builder: (context, child) {
        return MediaQuery(
          data: MediaQuery.of(context).copyWith(
            alwaysUse24HourFormat: TimeHelper.uses24HourFormat,
          ),
          child: child ?? const SizedBox.shrink(),
        );
      },
    );
  }

  Future<void> addReminderTime() async {
    final selectedTime = await pickCustomerTime(TimeOfDay.now());

    if (selectedTime == null) {
      return;
    }

    final updatedTimes = reminderTimes
        .map((time) => TimeHelper.timeToString(time))
        .toList();

    updatedTimes.add(TimeHelper.timeToString(selectedTime));
    updatedTimes.sort();

    setState(() {
      medication = medication.copyWith(
        reminderTimes: updatedTimes,
      );
    });

    await saveMedicationChanges();
  }

  Future<void> editReminderTime(int index) async {
    final times = reminderTimes;

    if (index < 0 || index >= times.length) {
      return;
    }

    final selectedTime = await pickCustomerTime(times[index]);

    if (selectedTime == null) {
      return;
    }

    final updatedTimes = times
        .map((time) => TimeHelper.timeToString(time))
        .toList();

    updatedTimes[index] = TimeHelper.timeToString(selectedTime);
    updatedTimes.sort();

    setState(() {
      medication = medication.copyWith(
        reminderTimes: updatedTimes,
      );
    });

    await saveMedicationChanges();
  }

  Future<void> deleteReminderTime(int index) async {
    final times = reminderTimes
        .map((time) => TimeHelper.timeToString(time))
        .toList();

    if (index < 0 || index >= times.length) {
      return;
    }

    times.removeAt(index);

    setState(() {
      medication = medication.copyWith(
        reminderTimes: times,
      );
    });

    await saveMedicationChanges();
  }

  Future<void> changeTimeFormat(String format) async {
    await TimeHelper.changeTimeFormat(format);

    if (mounted) {
      setState(() {});
    }
  }

  Future<void> editMedicationDetails() async {
    if (widget.medicationIndex < 0) {
      return;
    }

    final didSave = await Navigator.push(
      context,
      MaterialPageRoute(
        builder: (context) => MedicationDetailsPage(
          medication: medication,
          medicationIndex: widget.medicationIndex,
        ),
      ),
    );

    if (didSave == true) {
      final updatedMedications = await MedicationStorage.loadMedications();

      if (widget.medicationIndex >= 0 &&
          widget.medicationIndex < updatedMedications.length) {
        setState(() {
          medication = updatedMedications[widget.medicationIndex];
        });
      }
    }
  }

  String getNextDoseText() {
    final nextDose = TimeHelper.getNextDoseDateTime(reminderTimes);

    if (nextDose == null) {
      return "--";
    }

    return TimeHelper.formatDateTimeAsDisplayTime(nextDose);
  }

  int countStatus(String status) {
    int count = 0;

    for (final time in reminderTimes) {
      if (getDoseStatus(time) == status) {
        count++;
      }
    }

    return count;
  }

  double getTreatmentProgress() {
    final startDate = DateHelper.parseMedicationDate(medication.startDate);
    final endDate = DateHelper.parseMedicationDate(medication.endDate);

    if (startDate == null || endDate == null) {
      return 0;
    }

    return DateHelper.treatmentProgress(
      startDate: startDate,
      endDate: endDate,
    );
  }

  String getTreatmentText() {
    final language = AppLanguage.currentLanguage.value;
    final startDate = DateHelper.parseMedicationDate(medication.startDate);
    final endDate = DateHelper.parseMedicationDate(medication.endDate);

    if (startDate == null) {
      return language == "en"
          ? "No start date selected"
          : "Chưa chọn ngày bắt đầu";
    }

    if (endDate == null) {
      return language == "en"
          ? "Started ${DateHelper.displayMedicationDate(medication.startDate)} • No end date"
          : "Bắt đầu ${DateHelper.displayMedicationDate(medication.startDate)} • Không có ngày kết thúc";
    }

    final completedDays = DateHelper.treatmentCompletedDays(
      startDate: startDate,
      endDate: endDate,
    );

    final totalDays = DateHelper.treatmentTotalDays(
      startDate: startDate,
      endDate: endDate,
    );

    return language == "en"
        ? "$completedDays of $totalDays days completed"
        : "Đã hoàn thành $completedDays trên $totalDays ngày";
  }

  String getQuantityDisplayText() {
    final language = AppLanguage.currentLanguage.value;

    if (medication.quantity.trim().isEmpty) {
      return language == "en"
          ? "Quantity not entered"
          : "Chưa nhập số lượng";
    }

    final remaining = medication.remainingQuantity.trim().isEmpty
        ? medication.quantity
        : medication.remainingQuantity;

    return language == "en"
        ? "Remaining Qty: $remaining of ${medication.quantity}"
        : "Còn lại: $remaining trên ${medication.quantity}";
  }

  @override
  Widget build(BuildContext context) {
    final language = AppLanguage.currentLanguage.value;
    final times = reminderTimes;

    return Scaffold(
      body: AnimatedContainer(
        duration: const Duration(milliseconds: 350),
        curve: Curves.easeOutCubic,
        width: double.infinity,
        height: double.infinity,
        decoration: BoxDecoration(
          gradient: LinearGradient(
            colors: [
              AppTheme.lightColor,
              Colors.white,
            ],
            begin: Alignment.topCenter,
            end: Alignment.bottomCenter,
          ),
        ),
        child: SafeArea(
          child: SingleChildScrollView(
            padding: const EdgeInsets.all(24),
            child: Column(
              crossAxisAlignment: CrossAxisAlignment.start,
              children: [
                IconButton(
                  onPressed: () {
                    Navigator.pop(context, true);
                  },
                  icon: const Icon(Icons.arrow_back_ios),
                ),
                const SizedBox(height: 10),
                Text(
                  medication.name,
                  style: const TextStyle(
                    fontSize: 30,
                    fontWeight: FontWeight.bold,
                    color: Color(0xFF1E2A3A),
                  ),
                ),
                const SizedBox(height: 8),
                Text(
                  medication.dosage.trim().isEmpty
                      ? medication.instructions
                      : "${medication.dosage} • ${medication.instructions}",
                  style: const TextStyle(
                    fontSize: 16,
                    color: Color(0xFF667085),
                    height: 1.35,
                    fontWeight: FontWeight.w600,
                  ),
                ),
                const SizedBox(height: 18),
                TimeFormatChoiceCard(
                  selectedFormat: TimeHelper.currentTimeFormat.value,
                  onSelect: changeTimeFormat,
                ),
                const SizedBox(height: 18),
                DoseSummaryCard(
                  nextDoseText: getNextDoseText(),
                  takenCount: countStatus("taken"),
                  missedCount: countStatus("missed"),
                  dueCount: countStatus("due"),
                  upcomingCount: countStatus("upcoming"),
                ),
                const SizedBox(height: 18),
                QuantityRemainingCard(
                  text: getQuantityDisplayText(),
                ),
                const SizedBox(height: 18),
                TreatmentCard(
                  progress: getTreatmentProgress(),
                  treatmentText: getTreatmentText(),
                  startDate: medication.startDate,
                  endDate: medication.endDate,
                ),
                const SizedBox(height: 18),
                SmoothActionButton(
                  icon: Icons.edit_rounded,
                  label: language == "en"
                      ? "Edit Medication Details"
                      : "Sửa Thông Tin Thuốc",
                  outlined: true,
                  onPressed: editMedicationDetails,
                ),
                const SizedBox(height: 24),
                Row(
                  children: [
                    Expanded(
                      child: Text(
                        AppLanguage.text("reminderTimes"),
                        style: const TextStyle(
                          color: Color(0xFF1E2A3A),
                          fontSize: 22,
                          fontWeight: FontWeight.bold,
                        ),
                      ),
                    ),
                    TextButton.icon(
                      onPressed: addReminderTime,
                      icon: Icon(
                        Icons.add_rounded,
                        color: AppTheme.primaryColor,
                      ),
                      label: Text(
                        language == "en" ? "Add Time" : "Thêm Giờ",
                        style: TextStyle(
                          color: AppTheme.primaryColor,
                          fontWeight: FontWeight.bold,
                        ),
                      ),
                    ),
                  ],
                ),
                const SizedBox(height: 12),
                if (times.isEmpty)
                  EmptyReminderTimes(
                    onAddTime: addReminderTime,
                  )
                else
                  Column(
                    children: List.generate(times.length, (index) {
                      final time = times[index];
                      final status = getDoseStatus(time);

                      return DoseTimeCard(
                        time: time,
                        status: status,
                        onEdit: () {
                          editReminderTime(index);
                        },
                        onDelete: () {
                          deleteReminderTime(index);
                        },
                        onMarkTaken: () {
                          markDose(time, "taken");
                        },
                        onMarkMissed: () {
                          markDose(time, "missed");
                        },
                      );
                    }),
                  ),
                if (medication.notes.trim().isNotEmpty) ...[
                  const SizedBox(height: 20),
                  NotesCard(
                    notes: medication.notes,
                  ),
                ],
                const SizedBox(height: 24),
                const SafetyReminderCard(),
              ],
            ),
          ),
        ),
      ),
    );
  }
}

class TimeFormatChoiceCard extends StatelessWidget {
  final String selectedFormat;
  final Future<void> Function(String format) onSelect;

  const TimeFormatChoiceCard({
    super.key,
    required this.selectedFormat,
    required this.onSelect,
  });

  @override
  Widget build(BuildContext context) {
    final language = AppLanguage.currentLanguage.value;

    return Container(
      width: double.infinity,
      padding: const EdgeInsets.all(16),
      decoration: BoxDecoration(
        color: Colors.white,
        borderRadius: BorderRadius.circular(22),
        border: Border.all(
          color: AppTheme.primaryColor.withOpacity(0.16),
        ),
      ),
      child: Column(
        crossAxisAlignment: CrossAxisAlignment.start,
        children: [
          Text(
            language == "en" ? "Time Format" : "Kiểu Giờ",
            style: const TextStyle(
              color: Color(0xFF1E2A3A),
              fontSize: 17,
              fontWeight: FontWeight.bold,
            ),
          ),
          const SizedBox(height: 12),
          Row(
            children: [
              Expanded(
                child: TimeFormatButton(
                  label: "AM / PM",
                  example: "8:00 PM",
                  selected: selectedFormat == "ampm",
                  onTap: () {
                    onSelect("ampm");
                  },
                ),
              ),
              const SizedBox(width: 10),
              Expanded(
                child: TimeFormatButton(
                  label: "24-hour",
                  example: "20:00",
                  selected: selectedFormat == "24h",
                  onTap: () {
                    onSelect("24h");
                  },
                ),
              ),
            ],
          ),
        ],
      ),
    );
  }
}

class TimeFormatButton extends StatelessWidget {
  final String label;
  final String example;
  final bool selected;
  final VoidCallback onTap;

  const TimeFormatButton({
    super.key,
    required this.label,
    required this.example,
    required this.selected,
    required this.onTap,
  });

  @override
  Widget build(BuildContext context) {
    return Material(
      color: selected ? AppTheme.primaryColor : AppTheme.lightColor,
      borderRadius: BorderRadius.circular(18),
      child: InkWell(
        onTap: onTap,
        borderRadius: BorderRadius.circular(18),
        child: Container(
          padding: const EdgeInsets.symmetric(horizontal: 12, vertical: 12),
          decoration: BoxDecoration(
            borderRadius: BorderRadius.circular(18),
            border: Border.all(
              color: selected
                  ? AppTheme.primaryColor
                  : AppTheme.primaryColor.withOpacity(0.20),
            ),
          ),
          child: Column(
            children: [
              Text(
                label,
                style: TextStyle(
                  color: selected ? Colors.white : AppTheme.primaryColor,
                  fontWeight: FontWeight.bold,
                ),
              ),
              const SizedBox(height: 4),
              Text(
                example,
                style: TextStyle(
                  color:
                      selected ? Colors.white.withOpacity(0.85) : const Color(0xFF667085),
                  fontSize: 12,
                  fontWeight: FontWeight.w600,
                ),
              ),
            ],
          ),
        ),
      ),
    );
  }
}

class DoseSummaryCard extends StatelessWidget {
  final String nextDoseText;
  final int takenCount;
  final int missedCount;
  final int dueCount;
  final int upcomingCount;

  const DoseSummaryCard({
    super.key,
    required this.nextDoseText,
    required this.takenCount,
    required this.missedCount,
    required this.dueCount,
    required this.upcomingCount,
  });

  @override
  Widget build(BuildContext context) {
    final language = AppLanguage.currentLanguage.value;

    return AnimatedContainer(
      duration: const Duration(milliseconds: 260),
      curve: Curves.easeOutCubic,
      width: double.infinity,
      padding: const EdgeInsets.all(18),
      decoration: BoxDecoration(
        color: AppTheme.primaryColor,
        borderRadius: BorderRadius.circular(26),
        boxShadow: [
          BoxShadow(
            color: AppTheme.primaryColor.withOpacity(0.24),
            blurRadius: 20,
            offset: const Offset(0, 10),
          ),
        ],
      ),
      child: Column(
        children: [
          const Icon(
            Icons.notifications_active_rounded,
            color: Colors.white,
            size: 46,
          ),
          const SizedBox(height: 10),
          Text(
            "${language == "en" ? "Next dose" : "Liều tiếp theo"}: $nextDoseText",
            style: const TextStyle(
              color: Colors.white,
              fontSize: 23,
              fontWeight: FontWeight.bold,
            ),
          ),
          const SizedBox(height: 14),
          Wrap(
            spacing: 8,
            runSpacing: 8,
            alignment: WrapAlignment.center,
            children: [
              SummaryPill(
                text: "${language == "en" ? "Taken" : "Đã uống"} $takenCount",
                color: const Color(0xFF22C55E),
              ),

class TimeHelper {
  static ValueNotifier<String> currentTimeFormat = ValueNotifier("ampm");

  static const String _timeFormatKey = "selected_time_format";

  static Future<void> loadSavedTimeFormat() async {
    final prefs = await SharedPreferences.getInstance();
    final String savedFormat = prefs.getString(_timeFormatKey) ?? "ampm";

    if (savedFormat == "24h" || savedFormat == "ampm") {
      currentTimeFormat.value = savedFormat;
    } else {
      currentTimeFormat.value = "ampm";
    }
  }

  static Future<void> changeTimeFormat(String format) async {
    if (format != "24h" && format != "ampm") {
      return;
    }

    currentTimeFormat.value = format;

    final prefs = await SharedPreferences.getInstance();
    await prefs.setString(_timeFormatKey, format);
  }

  static bool get uses24HourFormat {
    return currentTimeFormat.value == "24h";
  }

  static String timeToString(TimeOfDay time) {
    final hour = time.hour.toString().padLeft(2, "0");
    final minute = time.minute.toString().padLeft(2, "0");

    return "$hour:$minute";
  }

  static TimeOfDay stringToTime(String value) {
    try {
      final parts = value.split(":");

      if (parts.length != 2) {
        return const TimeOfDay(hour: 8, minute: 0);
      }

      final hour = int.parse(parts[0]);
      final minute = int.parse(parts[1]);

      if (hour < 0 || hour > 23 || minute < 0 || minute > 59) {
        return const TimeOfDay(hour: 8, minute: 0);
      }

      return TimeOfDay(
        hour: hour,
        minute: minute,
      );
    } catch (e) {
      return const TimeOfDay(hour: 8, minute: 0);
    }
  }

  static String formatTimeForDisplay(TimeOfDay time) {
    if (uses24HourFormat) {
      return timeToString(time);
    }

    final period = time.hour >= 12 ? "PM" : "AM";
    int hour = time.hour % 12;

    if (hour == 0) {
      hour = 12;
    }

    final minute = time.minute.toString().padLeft(2, "0");

    return "$hour:$minute $period";
  }

  static String formatStoredTimeForDisplay(String value) {
    return formatTimeForDisplay(stringToTime(value));
  }

  static String formatDateTimeAsTime(DateTime dateTime) {
    final hour = dateTime.hour.toString().padLeft(2, "0");
    final minute = dateTime.minute.toString().padLeft(2, "0");

    return "$hour:$minute";
  }

  static String formatDateTimeAsDisplayTime(DateTime dateTime) {
    return formatTimeForDisplay(
      TimeOfDay(
        hour: dateTime.hour,
        minute: dateTime.minute,
      ),
    );
  }

  static List<TimeOfDay> generateReminderTimesFromInstructions(
    String instructions,
  ) {
    final text = instructions.toLowerCase();

    if (text.contains("twice daily") ||
        text.contains("twice a day") ||
        text.contains("2 times a day") ||
        text.contains("2 times per day") ||
        text.contains("2x") ||
        text.contains("bid") ||
        text.contains("every 12 hours") ||
        text.contains("q12h") ||
        text.contains("2 lần") ||
        text.contains("2 lan") ||
        text.contains("2 lan moi ngay") ||
        text.contains("2 lần mỗi ngày")) {
      return [
        const TimeOfDay(hour: 8, minute: 0),
        const TimeOfDay(hour: 20, minute: 0),
      ];
    }

    if (text.contains("3 times a day") ||
        text.contains("3 times per day") ||
        text.contains("three times daily") ||
        text.contains("3x") ||
        text.contains("tid") ||
        text.contains("every 8 hours") ||
        text.contains("q8h") ||
        text.contains("3 lần") ||
        text.contains("3 lan")) {
      return [
        const TimeOfDay(hour: 8, minute: 0),
        const TimeOfDay(hour: 14, minute: 0),
        const TimeOfDay(hour: 20, minute: 0),
      ];
    }

    if (text.contains("4 times a day") ||
        text.contains("4 times per day") ||
        text.contains("four times daily") ||
        text.contains("4x") ||
        text.contains("qid") ||
        text.contains("every 6 hours") ||
        text.contains("q6h") ||
        text.contains("4 lần") ||
        text.contains("4 lan")) {
      return [
        const TimeOfDay(hour: 6, minute: 0),
        const TimeOfDay(hour: 12, minute: 0),
        const TimeOfDay(hour: 18, minute: 0),
        const TimeOfDay(hour: 22, minute: 0),
      ];
    }

    if (text.contains("every night") ||
        text.contains("each night") ||
        text.contains("nightly") ||
        text.contains("bedtime") ||
        text.contains("at night") ||
        text.contains("mỗi tối") ||
        text.contains("moi toi") ||
        text.contains("mỗi đêm") ||
        text.contains("moi dem") ||
        text.contains("buổi tối") ||
        text.contains("buoi toi") ||
        text.contains("ngủ") ||
        text.contains("ngu")) {
      return [
        const TimeOfDay(hour: 21, minute: 0),
      ];
    }

    if (text.contains("every morning") ||
        text.contains("each morning") ||
        text.contains("morning") ||
        text.contains("mỗi sáng") ||
        text.contains("moi sang") ||
        text.contains("buổi sáng") ||
        text.contains("buoi sang") ||
        text.contains("sáng") ||
        text.contains("sang")) {
      return [
        const TimeOfDay(hour: 8, minute: 0),
      ];
    }

    if (text.contains("suppository") ||
        text.contains("suppositories") ||
        text.contains("rectal") ||
        text.contains("rectally") ||
        text.contains("đặt") ||
        text.contains("dat vao") ||
        text.contains("bệnh trĩ") ||
        text.contains("benh tri")) {
      return [
        const TimeOfDay(hour: 21, minute: 0),
      ];
    }

    final everyHourMatch = RegExp(
      r'every\s+(\d+)\s+hour',
      caseSensitive: false,
    ).firstMatch(text);

    if (everyHourMatch != null) {
      final intervalHours = int.tryParse(everyHourMatch.group(1) ?? "");

      if (intervalHours != null && intervalHours > 0 && intervalHours <= 24) {
        return generateTimesByInterval(intervalHours);
      }
    }

    final timesPerDayMatch = RegExp(
      r'(\d+)\s+times?\s+(a|per)\s+day',
      caseSensitive: false,
    ).firstMatch(text);

    if (timesPerDayMatch != null) {
      final timesPerDay = int.tryParse(timesPerDayMatch.group(1) ?? "");

      if (timesPerDay != null && timesPerDay > 0) {
        return generateTimesPerDay(timesPerDay);
      }
    }

    if (text.contains("once daily") ||
        text.contains("once a day") ||
        text.contains("1 time a day") ||
        text.contains("1 time per day") ||
        text.contains("daily") ||
        text.contains("mỗi ngày") ||
        text.contains("moi ngay")) {
      return [
        const TimeOfDay(hour: 8, minute: 0),
      ];
    }

    return [
      const TimeOfDay(hour: 8, minute: 0),
    ];
  }

  static List<TimeOfDay> generateTimesByInterval(int intervalHours) {
    final List<TimeOfDay> times = [];
    int hour = 8;

    for (int i = 0; i < 24; i += intervalHours) {
      times.add(
        TimeOfDay(
          hour: hour % 24,
          minute: 0,
        ),
      );

      hour += intervalHours;
    }

    return times;
  }

  static List<TimeOfDay> generateTimesPerDay(int timesPerDay) {
    if (timesPerDay <= 1) {
      return [
        const TimeOfDay(hour: 8, minute: 0),
      ];
    }

    if (timesPerDay == 2) {
      return [
        const TimeOfDay(hour: 8, minute: 0),
        const TimeOfDay(hour: 20, minute: 0),
      ];
    }

    if (timesPerDay == 3) {
      return [
        const TimeOfDay(hour: 8, minute: 0),
        const TimeOfDay(hour: 14, minute: 0),
        const TimeOfDay(hour: 20, minute: 0),
      ];
    }

    if (timesPerDay == 4) {
      return [
        const TimeOfDay(hour: 6, minute: 0),
        const TimeOfDay(hour: 12, minute: 0),
        const TimeOfDay(hour: 18, minute: 0),
        const TimeOfDay(hour: 22, minute: 0),
      ];
    }

    return generateTimesByInterval(24 ~/ timesPerDay);
  }

  static DateTime getDateTimeForToday(TimeOfDay time) {
    final now = DateTime.now();

    return DateTime(
      now.year,
      now.month,
      now.day,
      time.hour,
      time.minute,
    );
  }

  static DateTime? getNextDoseDateTime(List<TimeOfDay> reminderTimes) {
    if (reminderTimes.isEmpty) {
      return null;
    }

    final now = DateTime.now();
    final List<DateTime> doseDateTimes = [];

    for (final time in reminderTimes) {
      DateTime doseDateTime = getDateTimeForToday(time);

      if (doseDateTime.isBefore(now)) {
        doseDateTime = doseDateTime.add(const Duration(days: 1));
      }

      doseDateTimes.add(doseDateTime);
    }

    doseDateTimes.sort((a, b) => a.compareTo(b));

    return doseDateTimes.first;
  }

  static bool isSameTime(TimeOfDay a, TimeOfDay b) {
    return a.hour == b.hour && a.minute == b.minute;
  }

    if (isOpeningCamera || isOpeningGallery || isReadingText) {
      return;
    }

    setState(() {
      isOpeningCamera = true;
    });

    final language = AppLanguage.currentLanguage.value;

    try {
      final image = await imagePicker.pickImage(
        source: ImageSource.camera,
        imageQuality: 90,
        maxWidth: 1800,
      );

      if (image == null) {
        return;
      }

      setState(() {
        selectedImages.add(image);
      });

      await readTextFromAllImages();

      if (!mounted) return;

            language == "en"
                ? "Could not open camera. Please check camera permission."
                : "Không thể mở camera. Vui lòng kiểm tra quyền camera.",
          ),
        ),
      );
    } finally {
      if (mounted) {
        setState(() {
          isOpeningCamera = false;
        });
      }
    }
  }

    if (isOpeningCamera || isOpeningGallery || isReadingText) {
      return;
    }

    setState(() {
      isOpeningGallery = true;
    });

    final language = AppLanguage.currentLanguage.value;

    try {
      final images = await imagePicker.pickMultiImage(
        imageQuality: 90,
        maxWidth: 1800,
      );

        return;
      }

      setState(() {
        selectedImages.addAll(images);
      });

      await readTextFromAllImages();
    } catch (e) {
      if (!mounted) return;

      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(
          content: Text(
            language == "en"
                ? "Could not open photo library. Please check photo permission."
                : "Không thể mở thư viện ảnh. Vui lòng kiểm tra quyền ảnh.",
          ),
        ),
      );
    } finally {
      if (mounted) {
        setState(() {
          isOpeningGallery = false;
        });
      }
    }
  }

  void clearExtractedTextOnly() {
    scannedText = "";
    showRawText = false;
    nameController.clear();
    dosageController.clear();
    quantityController.clear();
    instructionsController.clear();
    notesController.clear();
  }

  Future<void> readTextFromAllImages() async {
    final language = AppLanguage.currentLanguage.value;

    if (selectedImages.isEmpty) {
      setState(() {
        clearExtractedTextOnly();
      });
      return;
    }

    setState(() {
      isReadingText = true;
      scannedText = "";
      showRawText = false;
    });

    final textRecognizer = TextRecognizer(
      script: TextRecognitionScript.latin,
    );

    try {
      final List<String> photoTexts = [];

      for (int i = 0; i < selectedImages.length; i++) {
        final inputImage = InputImage.fromFilePath(selectedImages[i].path);
        final recognizedText = await textRecognizer.processImage(inputImage);

        final cleanedText = recognizedText.text.trim();

        if (cleanedText.isNotEmpty) {
          photoTexts.add("Photo ${i + 1}:\n$cleanedText");
        }
      }

      final combinedText = photoTexts.join("\n\n");
      final extractedInfo = extractMedicationInfo(combinedText);

      if (!mounted) return;

      setState(() {
        scannedText = combinedText;
        nameController.text = extractedInfo.name;
        dosageController.text = extractedInfo.dosage;
        quantityController.text = extractedInfo.quantity;
        instructionsController.text = extractedInfo.instructions;
        notesController.text = extractedInfo.notes;
      });

      if (combinedText.trim().isEmpty && mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(
            content: Text(
              language == "en"
                  ? "No text found. Try taking clearer photos."
                  : "Không tìm thấy chữ. Hãy chụp rõ hơn.",
            ),
          ),
        );
      }
    } catch (e) {
      if (!mounted) return;

      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(
          content: Text(
            language == "en"
                ? "Could not read text from these photos. You can still enter details manually."
                : "Không thể đọc chữ từ các ảnh này. Bạn vẫn có thể nhập thủ công.",
          ),
        ),
      );
    } finally {
      await textRecognizer.close();

      if (mounted) {
        setState(() {
          isReadingText = false;
        });
      }
    }
  }

  MedicationExtractedInfo extractMedicationInfo(String text) {
    final lines = text
        .split("\n")
        .map((line) => line.trim())
        .where((line) => line.isNotEmpty)
        .toList();

    final knownResult = extractKnownMedicationResult(text, lines);

    if (knownResult != null) {
      return knownResult;
    }

    String name = extractKnownMedicationName(text);
    String dosage = extractKnownDosage(text);
    String quantity = extractQuantity(lines);
    String instructions = extractKnownInstructions(text);
    String notes = "";

    if (name.isEmpty) {
      int bestScore = -999;
      int bestIndex = -1;

      for (int i = 0; i < lines.length; i++) {
        final line = lines[i];

        if (shouldIgnoreForMedicationName(line)) {
          continue;
        }

        final cleanedName = cleanMedicationName(line);

        if (cleanedName.isEmpty) {
          continue;
        }

        final score = scoreMedicationNameCandidate(cleanedName);

        if (score > bestScore) {
          bestScore = score;
          bestIndex = i;
          name = cleanedName;
        }
      }

      if (name.isNotEmpty && dosage.isEmpty && bestIndex >= 0) {
        dosage = findDosageNearLine(lines, bestIndex);
      }
    }

    if (dosage.isEmpty) {
      dosage = extractDosageFromLines(lines);
    }

    if (instructions.isEmpty) {
      instructions = extractStrongInstructionFromText(text);
    }

    if (instructions.isEmpty) {
      instructions = extractBestInstructions(lines);
    }

    notes = extractNotes(lines);

    if (instructions.isEmpty) {
      instructions = "Follow prescription label directions.";
    }

    if (notes.isEmpty) {
      notes =
          "Scanned from prescription photo. Please double-check all information before saving.";
    } else {
      notes =
          "$notes\n\nScanned from prescription photo. Please double-check all information before saving.";
    }

    return MedicationExtractedInfo(
      name: name,
      dosage: dosage,
      quantity: quantity,
      instructions: instructions,
      notes: notes,
    );
  }

  String extractKnownDosage(String text) {
    final compact = normalizeTextForMatching(text);

    if (compact.contains("docusate") ||
        compact.contains("stoolsoftener") ||

        compact.contains("softener")) {

    }

    if ((compact.contains("hydrocortisone") ||
            compact.contains("hydrocortison") ||
            compact.contains("suppository") ||

    }

    if ((compact.contains("rosuvastatin") ||
            compact.contains("rosuvastatinca") ||
            compact.contains("crestor")) &&
        (compact.contains("5mg") || compact.contains("5"))) {
      return "5 MG";

  String extractDosageFromLines(List<String> lines) {
    final dosageRegExp = RegExp(
      r'\b\d+(\.\d+)?\s*(mg|mcg|g|gram|grams|ml|mL|units|unit|iu|IU|%)\b',
      caseSensitive: false,
    );

    for (final line in lines) {
      final match = dosageRegExp.firstMatch(line);

      if (match != null) {
        return normalizeDosage(match.group(0) ?? "");
      }
    }

    final noSpaceDosageRegExp = RegExp(
      r'\b\d+(\.\d+)?(mg|mcg|ml|iu)\b',
      caseSensitive: false,
    );

    for (final line in lines) {
      final match = noSpaceDosageRegExp.firstMatch(line);

      if (match != null) {
        final value = match.group(0) ?? "";

        return value
            .replaceAllMapped(
              RegExp(r'(\d+)([a-zA-Z]+)'),
              (match) => "${match.group(1)} ${match.group(2)}",
            )
            .toUpperCase();
      }
    }

    return "";
  }

  String extractQuantity(List<String> lines) {
    for (final line in lines) {
      final lower = line.toLowerCase();

      final qtyMatch = RegExp(
        r'\b(qty|quantity)\s*[:#]?\s*(\d{1,4})\b',
        caseSensitive: false,
      ).firstMatch(line);

      if (qtyMatch != null) {
        return qtyMatch.group(2) ?? "";
      }

      final qtyNoSpaceMatch = RegExp(
        r'\b(qty|quantity)(\d{1,4})\b',
        caseSensitive: false,
      ).firstMatch(line);

      if (qtyNoSpaceMatch != null) {
        return qtyNoSpaceMatch.group(2) ?? "";
      }

      final adultSuppMatch = RegExp(
        r'\b(\d{1,4})\s*adult\s*suppositor(?:y|ies)\b',
        caseSensitive: false,
      ).firstMatch(line);

      if (adultSuppMatch != null) {
        return adultSuppMatch.group(1) ?? "";
      }

      final packageSuppMatch = RegExp(
        r'\b(\d{1,4})\s*suppositor(?:y|ies)\b',
        caseSensitive: false,
      ).firstMatch(line);

      if (packageSuppMatch != null && !isInstructionLine(lower)) {
        return packageSuppMatch.group(1) ?? "";
      }

      final packageMatch = RegExp(
        r'\b(\d{1,4})\s*(tablet|tablets|tab|tabs|capsule|capsules|cap|caps|patch|patches|vial|vials|packet|packets|adult)\b',
        caseSensitive: false,
      ).firstMatch(line);

      if (packageMatch != null && !isInstructionLine(lower)) {
        return packageMatch.group(1) ?? "";
      }
    }

    return "";
  }

  int scoreMedicationNameCandidate(String value) {
    final lower = value.toLowerCase();
    int score = 0;

    if (value.length >= 5) {
      score += 2;
    }

    if (value == value.toUpperCase()) {
      score += 2;
    }

    final drugHints = [
      "docusate",
      "softener",
      "hydrocortisone",
      "hydrocortison",
      "acetate",
      "statin",
      "vastatin",
      "cillin",
      "cycline",
      "prazole",
      "sartan",
      "dipine",
      "olol",
      "formin",
      "mycin",
      "zole",
      "vir",
      "pril",
      "xetine",
      "zepam",
      "ca",
      "hcl",
      "er",
      "xr",
    ];

    for (final hint in drugHints) {
      if (lower.contains(hint)) {
        score += 10;
        break;
      }
    }

    final wordCount = value.split(" ").where((word) => word.isNotEmpty).length;

    if (wordCount >= 4) {
      score -= 5;
    }

    if (RegExp(r'\d').hasMatch(value)) {
      score -= 3;
    }

    if (lower.contains("duy") ||
        lower.contains("truong") ||
        lower.contains("anh") ||
        lower.contains("nguyen") ||
        lower.contains("tran") ||
        lower.contains(" le ")) {
      score -= 8;
    }

    return score;
  }

  String cleanMedicationName(String value) {
    String cleaned = value.trim();

    cleaned = cleaned.replaceAll(RegExp(r'[^a-zA-Z0-9\s\-]'), " ");
    cleaned = cleaned.replaceAll(RegExp(r'\s+'), " ").trim();

    final upper = cleaned.toUpperCase();

    if (upper.contains("DOCUSATE") || upper.contains("STOOL SOFTENER")) {
      return "DOCUSATE SODIUM";
    }

    if (upper.contains("ROSUVASTATIN")) {
      return "ROSUVASTATIN";
    }

    if (upper.contains("HYDROCORTISONE") ||
        upper.contains("HYDROCORTISON")) {
      return "HYDROCORTISONE ACETATE";
    }

    if (cleaned.toUpperCase().endsWith(" CAI")) {
      cleaned = cleaned.substring(0, cleaned.length - 1).trim();
    }

    if (cleaned.toUpperCase().endsWith(" CAL")) {
      cleaned = cleaned.substring(0, cleaned.length - 1).trim();
    }

    if (cleaned.length > 50) {
      cleaned = cleaned.substring(0, 50).trim();
    }

    return cleaned;
  }

  String findDosageNearLine(List<String> lines, int index) {
    final dosageRegExp = RegExp(
      r'\b\d+(\.\d+)?\s*(mg|mcg|g|gram|grams|ml|mL|units|unit|iu|IU|%)\b',
      caseSensitive: false,
    );

    final start = index - 4 < 0 ? 0 : index - 4;
    final end = index + 5 >= lines.length ? lines.length - 1 : index + 5;

    for (int i = start; i <= end; i++) {
      final match = dosageRegExp.firstMatch(lines[i]);

      if (match != null) {
        return normalizeDosage(match.group(0) ?? "");
      }
    }

    return "";
  }

  String extractStrongInstructionFromText(String text) {
    String lower = text.toLowerCase();

    lower = lower.replaceAll("\n", " ");
    lower = lower.replaceAll(RegExp(r'\s+'), " ");

    final compact = normalizeTextForMatching(lower);

    if (compact.contains("docusate") ||
        compact.contains("stoolsoftener") ||
        compact.contains("softener") ||
        compact.contains("taobon") ||
        compact.contains("táobón") ||
        compact.contains("constipation")) {
      return "Take 1 capsule twice daily as needed for constipation.";
    }

    if (compact.contains("takeonetableteverynight") ||
        compact.contains("takeonetableteverynightat") ||
        compact.contains("take1tableteverynight") ||
        compact.contains("takeonetableteverybedtime") ||
        compact.contains("take1tableteverybedtime")) {
      return "Take one tablet every night.";
    }

    final englishNightMatch = RegExp(
      r'take\s*(one|1)\s*tablet\s*(by mouth\s*)?(every|each)?\s*(night|bedtime)',
      caseSensitive: false,
    ).firstMatch(lower);

    if (englishNightMatch != null) {
      return "Take one tablet every night.";
    }

    if ((compact.contains("uong1vien") || compact.contains("uống1viên")) &&
        (compact.contains("moitoi") ||
            compact.contains("mỗitối") ||
            compact.contains("moidem") ||
            compact.contains("mỗidem") ||
            compact.contains("ngutri") ||
            compact.contains("ngủtrị"))) {

    }

    final vietnameseNightMatch = RegExp(
      r'(uống|uong)\s*1\s*(viên|vien).*?(mỗi|moi|ngủ|ngu|tối|toi)',
      caseSensitive: false,
    ).firstMatch(lower);

    if (vietnameseNightMatch != null) {
      return "Uống 1 viên mỗi tối.";
    }

    return "";
  }

  String extractBestInstructions(List<String> lines) {
    final instructionCandidates = <String>[];

    for (int i = 0; i < lines.length; i++) {
      final line = lines[i];
      final keywordIndex = instructionKeywordIndex(line);

      if (keywordIndex >= 0) {
        final parts = <String>[];

        final firstInstructionPart = line.substring(keywordIndex).trim();

        if (firstInstructionPart.isNotEmpty) {
          parts.add(firstInstructionPart);
        }

        for (int j = i + 1; j < lines.length && j <= i + 5; j++) {
          final currentLine = lines[j].trim();
          final currentLower = currentLine.toLowerCase();

          if (isInstructionStopLine(currentLower)) {
            break;
          }

          if (shouldIgnoreInstructionContinuation(currentLine)) {
            continue;
          }

          if (currentLine.isNotEmpty) {
            parts.add(currentLine);
          }
        }

        final candidate = cleanInstructionText(parts.join(" "));

        if (candidate.isNotEmpty) {
          instructionCandidates.add(candidate);
        }
      }
    }

    if (instructionCandidates.isEmpty) {
      return "";
    }

    instructionCandidates.sort(
      (a, b) => scoreInstructionCandidate(b).compareTo(
        scoreInstructionCandidate(a),
      ),
    );

    return instructionCandidates.first;
  }

  int instructionKeywordIndex(String line) {
    final lower = line.toLowerCase();

    final keywords = [
      "take",
      "uống",
      "uong",
      "use",
      "apply",
      "inject",
      "inhale",
      "chew",
      "dissolve",
      "insert",
      "đặt",
      "dat",
      "by mouth",
      "every",
      "daily",
      "twice",
      "bedtime",
      "rectally",
      "rectal",
    ];

    int bestIndex = -1;

    for (final keyword in keywords) {
      final index = lower.indexOf(keyword);

      if (index >= 0 && (bestIndex == -1 || index < bestIndex)) {
        bestIndex = index;
      }
    }

    return bestIndex;
  }

        lower.contains("prescriber") ||
        lower.contains("pharmacy") ||
        lower.contains("phone") ||
        lower.contains("date") ||
        lower.contains("cc#") ||
        RegExp(r'\d{6,}').hasMatch(lower);
  }

  int scoreInstructionCandidate(String value) {
    final lower = value.toLowerCase();
    int score = 0;

    if (lower.contains("take") ||
        lower.contains("uống") ||
        lower.contains("uong") ||
        lower.contains("insert") ||
        lower.contains("đặt") ||
        lower.contains("dat") ||
        lower.contains("rectally") ||
        lower.contains("rectal")) {
      score += 10;
    }

    if (lower.contains("tablet") ||
        lower.contains("capsule") ||
        lower.contains("supp") ||
        lower.contains("suppository") ||
        lower.contains("viên") ||
        lower.contains("vien")) {
      score += 6;
    }

    if (lower.contains("daily") ||
        lower.contains("every") ||
        lower.contains("twice") ||
        lower.contains("2 lần") ||
        lower.contains("2 lan") ||
        lower.contains("night") ||
        lower.contains("bedtime") ||
        lower.contains("mỗi") ||
        lower.contains("moi") ||
        lower.contains("ngủ") ||
        lower.contains("ngu") ||
        lower.contains("tối") ||
        lower.contains("toi")) {
      score += 6;
    }

    if (value.length > 20) {
      score += 4;
    }

    if (value.length > 140) {
      score -= 8;
    }

    if (lower.contains("patient") ||
        lower.contains("pharmacy") ||
        lower.contains("generic for") ||
        lower.contains("rx") ||
        lower.contains("ndc") ||
        lower.contains("remaining") ||
        lower.contains("qty") ||
        lower.contains("cc#") ||
        RegExp(r'\d{6,}').hasMatch(lower)) {
      score -= 15;
    }

    return score;
  }

  String cleanInstructionText(String value) {
    String cleaned = value.trim();

    cleaned = cleaned.replaceAll(RegExp(r'\s+'), " ");

    cleaned = cleaned.replaceAll(RegExp(r'CC#.*', caseSensitive: false), "");
    cleaned = cleaned.replaceAll(RegExp(r'QTY.*', caseSensitive: false), "");
    cleaned = cleaned.replaceAll(
      RegExp(r'REMAINING.*', caseSensitive: false),
      "",
    );
    cleaned = cleaned.replaceAll(RegExp(r'\d{6,}'), "");
    cleaned = cleaned.replaceAll(
      RegExp(r'Photo\s+\d+', caseSensitive: false),
      "",
    );

    cleaned = cleaned.replaceAll(
      RegExp(r'TAKEONE', caseSensitive: false),
      "TAKE ONE",
    );
    cleaned = cleaned.replaceAll(
      RegExp(r'TABLETEV', caseSensitive: false),
      "TABLET EV",
    );
    cleaned = cleaned.replaceAll(
      RegExp(r'TABLET EVERYNIGHT', caseSensitive: false),
      "TABLET EVERY NIGHT",
    );

    final takeIndex = cleaned.toLowerCase().indexOf("take");
    final insertIndex = cleaned.toLowerCase().indexOf("insert");
    final rectalIndex = cleaned.toLowerCase().indexOf("rectal");
    final datIndex = cleaned.toLowerCase().indexOf("đặt");
    final datNoAccentIndex = cleaned.toLowerCase().indexOf("dat");
    final uongIndex = cleaned.toLowerCase().indexOf("uống");
    final uongNoAccentIndex = cleaned.toLowerCase().indexOf("uong");

    final indexes = [
      takeIndex,
      insertIndex,
      rectalIndex,
      datIndex,
      datNoAccentIndex,
      uongIndex,
      uongNoAccentIndex,
    ].where((index) => index >= 0).toList();

    if (indexes.isNotEmpty) {
      indexes.sort();
      cleaned = cleaned.substring(indexes.first).trim();
    }

    cleaned = cleaned.replaceAll(RegExp(r'\s+'), " ").trim();

    if (cleaned.length > 180) {
      cleaned = cleaned.substring(0, 180).trim();
    }

    return cleaned;
  }

  String extractNotes(List<String> lines) {
    final noteParts = <String>[];

    for (int i = 0; i < lines.length; i++) {
      final line = lines[i];
      final lower = line.toLowerCase();

      if (lower.contains("warning") ||
          lower.contains("caution") ||
          lower.contains("avoid") ||
          lower.contains("do not") ||
          lower.contains("with food") ||
          lower.contains("water") ||
          lower.contains("plenty") ||
          lower.contains("may cause") ||
          lower.contains("finish all") ||
          lower.contains("alcohol") ||
          lower.contains("drowsy") ||
          lower.contains("antacid") ||
          lower.contains("aluminum") ||
          lower.contains("magnesium") ||
          lower.contains("for rectal use only") ||
          lower.contains("rectal use only")) {
        noteParts.add(line);

        for (int j = i + 1; j < lines.length && j <= i + 2; j++) {
          final nextLine = lines[j].trim();
          final nextLower = nextLine.toLowerCase();

          if (isInstructionStopLine(nextLower)) {
            break;
          }

          if (shouldIgnoreInstructionContinuation(nextLine)) {
            continue;
          }

          if (nextLine.isNotEmpty) {
            noteParts.add(nextLine);
          }
        }
      }
    }

    String notes = noteParts.join(" ");
    notes = notes.replaceAll(RegExp(r'\s+'), " ").trim();

    final lowerNotes = notes.toLowerCase();

    if (lowerNotes.contains("water") || lowerNotes.contains("plenty")) {
      notes = "Take with plenty of water.";
    }

    if (lowerNotes.contains("aluminum") ||
        lowerNotes.contains("magnesium") ||
        lowerNotes.contains("antacid")) {
      notes =
          "Do not take aluminum or magnesium antacids within 2 hours of taking this medication.";
    }

    if (lowerNotes.contains("rectal")) {
      notes = "For rectal use only.";
    }

    if (notes.length > 180) {
      notes = notes.substring(0, 180).trim();
    }

    return notes;
  }

    }
  }

  void clearAllImages() {

      selectedImages.clear();

    });

  String dateToString(DateTime date) {
    final year = date.year.toString().padLeft(4, "0");
    final month = date.month.toString().padLeft(2, "0");
    final day = date.day.toString().padLeft(2, "0");

    return "$year-$month-$day";
  }

class MedicationDetailsPage extends StatefulWidget {
  final Medication? medication;
  final int? medicationIndex;
  final bool returnToPreviousPageAfterSave;

  const MedicationDetailsPage({
    super.key,
    this.medication,
    this.medicationIndex,
    this.returnToPreviousPageAfterSave = true,
  });

  @override
  State<MedicationDetailsPage> createState() => _MedicationDetailsPageState();
}

class _MedicationDetailsPageState extends State<MedicationDetailsPage> {
  final TextEditingController nameController = TextEditingController();
  final TextEditingController dosageController = TextEditingController();
  final TextEditingController quantityController = TextEditingController();
  final TextEditingController instructionsController = TextEditingController();
  final TextEditingController notesController = TextEditingController();

  bool isSaving = false;

  String startDate = "";
  String endDate = "";

  bool get isEditing => widget.medication != null;

  @override
  void initState() {
    super.initState();

    final medication = widget.medication;

    if (medication != null) {
      nameController.text = medication.name;
      dosageController.text = medication.dosage;
      quantityController.text = medication.quantity;
      instructionsController.text = medication.instructions;
      notesController.text = medication.notes;
      startDate = medication.startDate;
      endDate = medication.endDate;
    } else {
      startDate = DateHelper.todayString();
    }
  }

  @override
  void dispose() {
    nameController.dispose();
    dosageController.dispose();
    quantityController.dispose();
    instructionsController.dispose();
    notesController.dispose();
    super.dispose();
  }

  void useExampleMedication() {
    setState(() {
      nameController.text = "Rosuvastatin";
      dosageController.text = "5 MG";
      quantityController.text = "30";
      instructionsController.text = "Take one tablet every night.";
      notesController.text =
          "Do not take aluminum or magnesium antacids within 2 hours of taking this medication.";
      startDate = DateHelper.todayString();
      endDate = "";
    });
  }

  Future<void> pickStartDate() async {
    final initialDate =
        DateHelper.parseMedicationDate(startDate) ?? DateTime.now();

    final selectedDate = await showDatePicker(
      context: context,
      initialDate: initialDate,
      firstDate: DateTime(2020),
      lastDate: DateTime(2100),
    );

    if (selectedDate == null) {
      return;
    }

    setState(() {
      startDate = DateHelper.dateToString(selectedDate);

      final parsedEndDate = DateHelper.parseMedicationDate(endDate);

      if (parsedEndDate != null && parsedEndDate.isBefore(selectedDate)) {
        endDate = "";
      }
    });
  }

  Future<void> pickEndDate() async {
    final parsedStartDate =
        DateHelper.parseMedicationDate(startDate) ?? DateTime.now();

    final initialDate = DateHelper.parseMedicationDate(endDate) ??
        parsedStartDate.add(const Duration(days: 7));

    final selectedDate = await showDatePicker(
      context: context,
      initialDate: initialDate.isBefore(parsedStartDate)
          ? parsedStartDate
          : initialDate,
      firstDate: parsedStartDate,
      lastDate: DateTime(2100),
    );

    if (selectedDate == null) {
      return;
    }

    setState(() {
      endDate = DateHelper.dateToString(selectedDate);
    });
  }

  void clearEndDate() {
    setState(() {
      endDate = "";
    });
  }

  int parseNumber(String value) {
    final match = RegExp(r'\d+').firstMatch(value);

    if (match == null) {
      return 0;
    }

    return int.tryParse(match.group(0) ?? "") ?? 0;
  }

  String calculateEndDateFromQuantity({
    required String quantity,
    required List<String> reminderTimes,
  }) {
    final start = DateHelper.parseMedicationDate(startDate) ?? DateTime.now();
    final qty = parseNumber(quantity);

    if (qty <= 0) {
      return "";
    }

    final dosesPerDay = reminderTimes.isEmpty ? 1 : reminderTimes.length;
    final totalDays = (qty / dosesPerDay).ceil();

    if (totalDays <= 0) {
      return "";
    }

    final calculatedEndDate = start.add(
      Duration(days: totalDays - 1),
    );

    return DateHelper.dateToString(calculatedEndDate);
  }

  Future<void> saveMedication() async {
    if (isSaving) {
      return;
    }

    final language = AppLanguage.currentLanguage.value;

    final name = nameController.text.trim();
    final dosage = dosageController.text.trim();
    final quantity = quantityController.text.trim();
    final instructions = instructionsController.text.trim();
    final notes = notesController.text.trim();

    if (name.isEmpty || dosage.isEmpty || instructions.isEmpty) {
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(
          content: Text(AppLanguage.text("fillAllFields")),
        ),
      );
      return;
    }

    setState(() {
      isSaving = true;
    });

    try {
      final oldMedication = widget.medication;
      final oldInstructions = oldMedication?.instructions.trim() ?? "";
      final oldQuantity = oldMedication?.quantity.trim() ?? "";
      final instructionsChanged = oldInstructions != instructions;

      List<String> reminderTimes = oldMedication?.reminderTimes ?? [];

      if (reminderTimes.isEmpty || instructionsChanged) {
        reminderTimes = TimeHelper.generateReminderTimesFromInstructions(
          instructions,
        ).map((time) => TimeHelper.timeToString(time)).toList();
      }

      String finalEndDate = endDate;

      if (finalEndDate.isEmpty) {
        final calculatedEndDate = calculateEndDateFromQuantity(
          quantity: quantity,
          reminderTimes: reminderTimes,
        );

        if (calculatedEndDate.isNotEmpty) {
          finalEndDate = calculatedEndDate;
        }
      }

      String remainingQuantity = oldMedication?.remainingQuantity ?? quantity;

      if (!isEditing) {
        remainingQuantity = quantity;
      } else if (oldQuantity != quantity) {
        remainingQuantity = quantity;
      } else if (remainingQuantity.trim().isEmpty) {
        remainingQuantity = quantity;
      }

      final savedMedication = Medication(
        name: name,
        dosage: dosage,
        quantity: quantity,
        remainingQuantity: remainingQuantity,
        instructions: instructions,
        notes: notes,
        reminderTimes: reminderTimes,
        doseStatus: instructionsChanged
            ? "notTakenYet"
            : oldMedication?.doseStatus ?? "notTakenYet",
        doseStatusDate:
            instructionsChanged ? "" : oldMedication?.doseStatusDate ?? "",
        doseRecords:
            instructionsChanged ? const {} : oldMedication?.doseRecords ?? {},
        startDate: startDate.isEmpty ? DateHelper.todayString() : startDate,
        endDate: finalEndDate,
      );

      if (widget.medicationIndex != null) {
        await MedicationStorage.updateMedication(
          widget.medicationIndex!,
          savedMedication,
        );
      } else {
        await MedicationStorage.saveMedication(savedMedication);
      }

      if (!mounted) return;

      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(
          content: Text(
            language == "en"
                ? "Medication saved successfully."
                : "Đã lưu thuốc thành công.",
          ),
        ),
      );

      Navigator.pop(context, true);
    } catch (e) {
      if (!mounted) return;

      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(
          content: Text(
            language == "en"
                ? "Could not save medication."
                : "Không thể lưu thuốc.",
          ),
        ),
      );
    } finally {
      if (mounted) {
        setState(() {
          isSaving = false;
        });
      }
    }
  }

  @override
  Widget build(BuildContext context) {
    final language = AppLanguage.currentLanguage.value;

    return Scaffold(
      body: AnimatedContainer(
        duration: const Duration(milliseconds: 350),
        curve: Curves.easeOutCubic,
        width: double.infinity,
        height: double.infinity,
        decoration: BoxDecoration(
          gradient: LinearGradient(
            colors: [
              AppTheme.lightColor,
              Colors.white,
            ],
            begin: Alignment.topCenter,
            end: Alignment.bottomCenter,
          ),
        ),
        child: SafeArea(
          child: SingleChildScrollView(
            padding: const EdgeInsets.all(24),
            child: Column(
              crossAxisAlignment: CrossAxisAlignment.start,
              children: [
                IconButton(
                  onPressed: () {
                    Navigator.pop(context);
                  },
                  icon: const Icon(Icons.arrow_back_ios),
                ),
                const SizedBox(height: 10),
                Text(
                  AppLanguage.text("medicationDetails"),
                  style: const TextStyle(
                    fontSize: 30,
                    fontWeight: FontWeight.bold,
                    color: Color(0xFF1E2A3A),
                  ),
                ),
                const SizedBox(height: 8),
                Text(
                  isEditing
                      ? (language == "en"
                          ? "Update the medication information carefully."
                          : "Cập nhật thông tin thuốc cẩn thận.")
                      : (language == "en"
                          ? "Enter medication information carefully."
                          : "Nhập thông tin thuốc cẩn thận."),
                  style: const TextStyle(
                    fontSize: 16,
                    color: Color(0xFF667085),
                    height: 1.35,
                  ),
                ),
                const SizedBox(height: 24),
                MedicationFormCard(
                  nameController: nameController,
                  dosageController: dosageController,
                  quantityController: quantityController,
                  instructionsController: instructionsController,
                  notesController: notesController,
                  startDate: startDate,
                  endDate: endDate,
                  onPickStartDate: pickStartDate,
                  onPickEndDate: pickEndDate,
                  onClearEndDate: clearEndDate,
                ),
                const SizedBox(height: 18),
                if (!isEditing) ...[
                  SmoothActionButton(
                    icon: Icons.science_rounded,
                    label: language == "en"
                        ? "Use Example for Testing"
                        : "Dùng Ví Dụ Để Kiểm Tra",
                    outlined: true,
                    onPressed: useExampleMedication,
                  ),
                  const SizedBox(height: 14),
                ],
                SmoothActionButton(
                  icon: Icons.save_rounded,
                  label: AppLanguage.text("saveMedication"),
                  onPressed: saveMedication,
                  isLoading: isSaving,
                ),
              ],
            ),
          ),
        ),
      ),
    );
  }
}

  const MedicationListPage({super.key});

  @override
  State<MedicationListPage> createState() => _MedicationListPageState();
}

class _MedicationListPageState extends State<MedicationListPage> {
  final TextEditingController searchController = TextEditingController();

  List<Medication> medications = [];
  List<MedicationListItem> visibleItems = [];

  bool isLoading = true;
  String selectedFilter = "all";

  @override
  void initState() {
    super.initState();
    loadMedications();

    searchController.addListener(() {
      applySearchAndFilter();
    });
  }

  @override
  void dispose() {
    searchController.dispose();
    super.dispose();
  }

  Future<void> loadMedications() async {
    setState(() {
      isLoading = true;
    });

    final loadedMedications = await MedicationStorage.loadMedications();

    if (!mounted) return;

    setState(() {
      medications = loadedMedications;
      isLoading = false;
    });

    applySearchAndFilter();
  }

  void applySearchAndFilter() {
    final searchText = searchController.text.trim().toLowerCase();

    final List<MedicationListItem> items = [];

    for (int i = 0; i < medications.length; i++) {
      final medication = medications[i];

      final matchesSearch = medication.name.toLowerCase().contains(searchText) ||
          medication.dosage.toLowerCase().contains(searchText) ||
          medication.quantity.toLowerCase().contains(searchText) ||
          medication.remainingQuantity.toLowerCase().contains(searchText) ||
          medication.instructions.toLowerCase().contains(searchText) ||
          medication.notes.toLowerCase().contains(searchText);

      final matchesFilter = selectedFilter == "all" ||
          (selectedFilter == "active" && isMedicationActive(medication)) ||
          (selectedFilter == "upcoming" && isMedicationUpcoming(medication)) ||
          (selectedFilter == "ended" && isMedicationEnded(medication));

      if (matchesSearch && matchesFilter) {
        items.add(
          MedicationListItem(
            medication: medication,
            originalIndex: i,
          ),
        );
      }
    }

    setState(() {
      visibleItems = items;
    });
  }

  bool isMedicationActive(Medication medication) {
    final startDate = DateHelper.parseMedicationDate(medication.startDate);
    final endDate = DateHelper.parseMedicationDate(medication.endDate);
    final today = DateHelper.dateOnly(DateTime.now());

    if (startDate != null && today.isBefore(DateHelper.dateOnly(startDate))) {
      return false;
    }

    if (endDate != null && today.isAfter(DateHelper.dateOnly(endDate))) {
      return false;
    }

    return true;
  }

  bool isMedicationUpcoming(Medication medication) {
    final startDate = DateHelper.parseMedicationDate(medication.startDate);
    final today = DateHelper.dateOnly(DateTime.now());

    if (startDate == null) {
      return false;
    }

    return DateHelper.dateOnly(startDate).isAfter(today);
  }

  bool isMedicationEnded(Medication medication) {
    final endDate = DateHelper.parseMedicationDate(medication.endDate);
    final today = DateHelper.dateOnly(DateTime.now());

    if (endDate == null) {
      return false;
    }

    return DateHelper.dateOnly(endDate).isBefore(today);
  }

  Future<void> openMedicationReminder(
    Medication medication,
    int originalIndex,
  ) async {
    final didChange = await Navigator.push(
      context,
      MaterialPageRoute(
        builder: (context) => ReminderPage(
          medication: medication,
          medicationIndex: originalIndex,
        ),
      ),
    );

    if (didChange == true) {
      await loadMedications();
    } else {
      await loadMedications();
    }
  }

  Future<void> editMedication(
    Medication medication,
    int originalIndex,
  ) async {
    final didSave = await Navigator.push(
      context,
      MaterialPageRoute(
        builder: (context) => MedicationDetailsPage(
          medication: medication,
          medicationIndex: originalIndex,
        ),
      ),
    );

    if (didSave == true) {
      await loadMedications();
    }
  }

  Future<void> confirmDeleteMedication(
    Medication medication,
    int originalIndex,
  ) async {
    final language = AppLanguage.currentLanguage.value;

    final shouldDelete = await showDialog<bool>(
      context: context,
      builder: (context) {
        return AlertDialog(
          title: Text(
            language == "en" ? "Delete Medication?" : "Xoá Thuốc?",
          ),
          content: Text(
            language == "en"
                ? "Are you sure you want to delete ${medication.name}?"
                : "Bạn có chắc muốn xoá ${medication.name} không?",
          ),
          actions: [
            TextButton(
              onPressed: () {
                Navigator.pop(context, false);
              },
              child: Text(
                language == "en" ? "Cancel" : "Huỷ",
              ),
            ),
            TextButton(
              onPressed: () {
                Navigator.pop(context, true);
              },
              child: Text(
                language == "en" ? "Delete" : "Xoá",
                style: const TextStyle(
                  color: Color(0xFFEF4444),
                  fontWeight: FontWeight.bold,
                ),
              ),
            ),
          ],
        );
      },
    );

    if (shouldDelete != true) {
      return;
    }

    await MedicationStorage.deleteMedication(originalIndex);
    await loadMedications();

    if (!mounted) return;

    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(
        content: Text(
          language == "en" ? "Medication deleted." : "Đã xoá thuốc.",
        ),
        action: SnackBarAction(
          label: language == "en" ? "Undo" : "Hoàn tác",
          onPressed: () async {
            final updatedMedications = await MedicationStorage.loadMedications();

            if (originalIndex >= 0 &&
                originalIndex <= updatedMedications.length) {
              updatedMedications.insert(originalIndex, medication);
            } else {
              updatedMedications.add(medication);
            }

            await MedicationStorage.saveMedications(updatedMedications);
            await loadMedications();
          },
        ),
      ),
    );
  }

  String getNextDoseText(Medication medication) {
    final times = getReminderTimes(medication);

    if (times.isEmpty) {
      return "--";
    }

    final nextDose = TimeHelper.getNextDoseDateTime(times);

    if (nextDose == null) {
      return "--";
    }

    return TimeHelper.formatDateTimeAsDisplayTime(nextDose);
  }

  List<TimeOfDay> getReminderTimes(Medication medication) {
    if (medication.reminderTimes.isNotEmpty) {
      return medication.reminderTimes
          .map((time) => TimeHelper.stringToTime(time))
          .toList();
    }

    return TimeHelper.generateReminderTimesFromInstructions(
      medication.instructions,
    );
  }

  String getTodayStatusSummary(Medication medication) {
    final language = AppLanguage.currentLanguage.value;
    final today = DateHelper.todayString();
    final times = getReminderTimes(medication);

    int takenCount = 0;
    int missedCount = 0;

    for (final time in times) {
      final key = "$today|${TimeHelper.timeToString(time)}";
      final status = medication.doseRecords[key] ?? "";

      if (status == "taken") {
        takenCount++;
      }

      if (status == "missed") {
        missedCount++;
      }
    }

    if (takenCount == 0 && missedCount == 0) {
      return language == "en" ? "No dose marked today" : "Chưa đánh dấu hôm nay";
    }

    return language == "en"
        ? "$takenCount taken • $missedCount missed"
        : "$takenCount đã uống • $missedCount bỏ lỡ";
  }

class MedicationCard extends StatelessWidget {
  final Medication medication;
  final String nextDoseText;
  final String todayStatusText;
  final bool isActive;
  final bool isUpcoming;
  final bool isEnded;
  final VoidCallback onTap;
  final VoidCallback onEdit;
  final VoidCallback onDelete;

  const MedicationCard({
    super.key,
    required this.medication,
    required this.nextDoseText,
    required this.todayStatusText,
    required this.isActive,
    required this.isUpcoming,
    required this.isEnded,
    required this.onTap,
    required this.onEdit,
    required this.onDelete,
  });

  String statusText() {
    final language = AppLanguage.currentLanguage.value;

    if (isEnded) {
      return language == "en" ? "Ended" : "Đã kết thúc";
    }

    if (isUpcoming) {
      return language == "en" ? "Upcoming" : "Sắp bắt đầu";
    }

    return language == "en" ? "Active" : "Đang dùng";
  }

  Color statusColor() {
    if (isEnded) {
      return const Color(0xFFEF4444);
    }

    if (isUpcoming) {
      return const Color(0xFFF59E0B);
    }

    return const Color(0xFF22C55E);
  }

  String quantityLine() {
    if (medication.quantity.trim().isEmpty) {
      return medication.dosage;
    }

    final remaining = medication.remainingQuantity.trim().isEmpty
        ? medication.quantity
        : medication.remainingQuantity;

    return "${medication.dosage} • Remaining $remaining/${medication.quantity}";
  }

    final language = AppLanguage.currentLanguage.value;
